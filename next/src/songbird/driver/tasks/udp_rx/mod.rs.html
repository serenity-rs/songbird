53    async fn run(&mut self, interconnect: &mut Interconnect) {
54        let mut cleanup_time = Instant::now();
55        let mut playout_time = Instant::now() + TIMESTEP_LENGTH;
56        let mut byte_dest: Option<BytesMut> = None;
57
58        loop {
59            if byte_dest.is_none() {
60                byte_dest = Some(BytesMut::zeroed(VOICE_PACKET_MAX));
61            }
62
63            select! {
64                Ok((len, _addr)) = self.udp_socket.recv_from(byte_dest.as_mut().unwrap()) => {
65                    let mut pkt = byte_dest.take().unwrap();
66                    pkt.truncate(len);
67
68                    self.process_udp_message(interconnect, pkt).await;
69                },
70                msg = self.rx.recv_async() => {
71                    match msg {
72                        Ok(UdpRxMessage::ReplaceInterconnect(i)) => {
73                            *interconnect = i;
74                        },
75                        Ok(UdpRxMessage::SetConfig(new_config)) => {
76                            if let DecodeMode::Decode(old_config) = &mut self.config.decode_mode {
77                                if *old_config != new_config {
78                                    *old_config = new_config;
79                                    self.decoder_map.values_mut().for_each(|v| v.reconfigure_decoder(new_config));
80                                }
81                            }
82                        },
83                        Err(flume::RecvError::Disconnected) => break,
84                    }
85                },
86                () = tokio::time::sleep_until(playout_time) => {
87                    let mut tick = VoiceTick {
88                        speaking: HashMap::new(),
89                        silent: HashSet::new(),
90                    };
91
92                    for (ssrc, state) in &mut self.decoder_map {
93                        match state.get_voice_tick(&self.config) {
94                            Ok(Some(data)) => {
95                                tick.speaking.insert(*ssrc, data);
96                            },
97                            Ok(None) => {
98                                if !state.disconnected {
99                                    tick.silent.insert(*ssrc);
100                                }
101                            },
102                            Err(e) => {
103                                warn!("Decode error for SSRC {ssrc}: {e:?}");
104                                tick.silent.insert(*ssrc);
105                            },
106                        }
107                    }
108
109                    playout_time += TIMESTEP_LENGTH;
110
111                    drop(interconnect.events.send(EventMessage::FireCoreEvent(CoreContext::VoiceTick(tick))));
112                },
113                () = tokio::time::sleep_until(cleanup_time) => {
114                    // periodic cleanup.
115                    let now = Instant::now();
116
117                    // check ssrc map to see if the WS task has informed us of any disconnects.
118                    loop {
119                        // This is structured in an odd way to prevent deadlocks.
120                        // while-let seemed to keep the dashmap iter() alive for block scope, rather than
121                        // just the initialiser.
122                        let id = {
123                            if let Some(id) = self.ssrc_signalling.disconnected_users.iter().next().map(|v| *v.key()) {
124                                id
125                            } else {
126                                break;
127                            }
128                        };
129
130                        _ = self.ssrc_signalling.disconnected_users.remove(&id);
131                        if let Some((_, ssrc)) = self.ssrc_signalling.user_ssrc_map.remove(&id) {
132                            let _ = self.ssrc_signalling.ssrc_user_map.remove(&ssrc);
133
134                            if let Some(state) = self.decoder_map.get_mut(&ssrc) {
135                                // don't cleanup immediately: leave for later cycle
136                                // this is key with reorder/jitter buffers where we may
137                                // still need to decode post disconnect for ~0.2s.
138                                state.prune_time = now + Duration::from_secs(1);
139                                state.disconnected = true;
140                            }
141                        }
142                    }
143
144                    // now remove all dead ssrcs.
145                    self.decoder_map.retain(|_, v| v.prune_time > now);
146
147                    cleanup_time = now + Duration::from_secs(5);
148                },
149            }
150        }
151    }

153    async fn process_udp_message(&mut self, interconnect: &Interconnect, mut packet: BytesMut) {
154        // NOTE: errors here (and in general for UDP) are not fatal to the connection.
155        // Panics should be avoided due to adversarial nature of rx'd packets,
156        // but correct handling should not prompt a reconnect.
157        //
158        // For simplicity, if the event task fails then we nominate the mixing thread
159        // to rebuild their context etc. (hence, the `let _ =` statements.), as it will
160        // try to make contact every 20ms.
161        let crypto_mode = self.crypto_mode;
162
163        match demux::demux_mut(packet.as_mut()) {
164            DemuxedMut::Rtp(mut rtp) => {
165                if !rtp_valid(&rtp.to_immutable()) {
166                    error!("Illegal RTP message received.");
167                    return;
168                }
169
170                let mut packet_data = if self.config.decode_mode.should_decrypt() {
171                    let out = self.cipher.decrypt_rtp_in_place(&mut rtp).map(|(s, t)| {
172                        if rtp.get_padding() != 0 {
173                            let payload = rtp.payload();
174                            let payload_length = payload.len();
175                            let padding_count = payload[payload_length - t - 1] as usize;
176
177                            (s, t + padding_count, true)
178                        } else {
179                            (s, t, true)
180                        }
181                    });
182
183                    if let Err(ref e) = out {
184                        warn!("RTP decryption failed: {:?}", e);
185                    }
186
187                    out.ok()
188                } else {
189                    None
190                };
191
192                if let Some((rtp_body_start, rtp_body_tail, decrypted)) = packet_data {
193                    let ssrc = rtp.get_ssrc();
194                    let payload = rtp.payload_mut();
195                    let payload_length = payload.len();
196                    let body = &mut payload[rtp_body_start..payload_length - rtp_body_tail];
197                    let body_length = body.len();
198
199                    // If the packet is transport-decrypted, DAVE is enabled, and the packet
200                    // looks encrypted: https://daveprotocol.com/#protocol-frame-check
201                    // The packet must be at least 11 bytes consisting of:
202                    // - 8 byte truncated AES128-GCM authentication tag
203                    // - 1 byte protocol supplemental data size
204                    // - 2 byte magic marker
205                    if decrypted
206                        && self.dave_protocol_version.load(Ordering::Relaxed) != 0
207                        && body_length >= 11
208                        && body[body_length - DAVE_MAGIC_MARKER.len()..body_length]
209                            == DAVE_MAGIC_MARKER
210                    {
211                        // Silently drop encrypted packets if it's not possible to decrypt them for
212                        // one reason or another; otherwise there'd be error logs for trying to decode
213                        // Opus
214                        let Some(user_id) = self.ssrc_signalling.ssrc_user_map.get(&ssrc) else {
215                            return;
216                        };
217                        let Some(ref mut dave_session) = *self.dave_session.write().unwrap() else {
218                            return;
219                        };
220
221                        if !dave_session.is_ready() {
222                            return;
223                        }
224
225                        let result = dave_session.decrypt(user_id.0, davey::MediaType::AUDIO, body);
226
227                        match result {
228                            Ok(decrypted_body) => {
229                                packet_data = Some((
230                                    rtp_body_start,
231                                    rtp_body_tail + (body.len() - decrypted_body.len()),
232                                    decrypted,
233                                ));
234                                body[..decrypted_body.len()].copy_from_slice(&decrypted_body);
235                            },
236                            Err(davey::errors::DecryptError::NoDecryptorForUser)
237                            | Err(davey::errors::DecryptError::DecryptionFailed(
238                                davey::errors::DecryptorDecryptError::NoValidCryptorFound {
239                                    ..
240                                },
241                            )) => {
242                                // Silently drop encrypted packets for users whose ratchets are not configured yet.
243                                return;
244                            },
245                            Err(e) => {
246                                error!(error = ?e, "DAVE decryption failed");
247                                return;
248                            },
249                        }
250                    }
251                }
252
253                let rtp = rtp.to_immutable();
254                let (rtp_body_start, rtp_body_tail, decrypted) = packet_data.unwrap_or_else(|| {
255                    (
256                        crypto_mode.payload_prefix_len(),
257                        crypto_mode.payload_suffix_len(),
258                        false,
259                    )
260                });
261
262                let entry = self
263                    .decoder_map
264                    .entry(rtp.get_ssrc())
265                    .or_insert_with(|| SsrcState::new(&rtp, crypto_mode, &self.config));
266
267                // Only do this on RTP, rather than RTCP -- this pins decoder state liveness
268                // to *speech* rather than just presence.
269                entry.refresh_timer(self.config.decode_state_timeout.into());
270
271                let store_pkt = StoredPacket {
272                    packet: packet.freeze(),
273                    decrypted,
274                };
275                let packet = store_pkt.packet.clone();
276                entry.store_packet(store_pkt, &self.config);
277
278                drop(interconnect.events.send(EventMessage::FireCoreEvent(
279                    CoreContext::RtpPacket(InternalRtpPacket {
280                        packet,
281                        payload_offset: rtp_body_start,
282                        payload_end_pad: rtp_body_tail,
283                    }),
284                )));
285            },
286            DemuxedMut::Rtcp(mut rtcp) => {
287                let packet_data = if self.config.decode_mode.should_decrypt() {
288                    let out = self.cipher.decrypt_rtcp_in_place(&mut rtcp);
289
290                    if let Err(ref e) = out {
291                        warn!("RTCP decryption failed: {:?}", e);
292                    }
293
294                    out.ok()
295                } else {
296                    None
297                };
298
299                let (start, tail) = packet_data.unwrap_or_else(|| {
300                    (
301                        crypto_mode.payload_prefix_len(),
302                        crypto_mode.payload_suffix_len(),
303                    )
304                });
305
306                drop(interconnect.events.send(EventMessage::FireCoreEvent(
307                    CoreContext::RtcpPacket(InternalRtcpPacket {
308                        packet: packet.freeze(),
309                        payload_offset: start,
310                        payload_end_pad: tail,
311                    }),
312                )));
313            },
314            DemuxedMut::FailedParse(t) => {
315                warn!("Failed to parse message of type {:?}.", t);
316            },
317            DemuxedMut::TooSmall => {
318                warn!("Illegal UDP packet from voice server.");
319            },
320        }
321    }
322}

324#[instrument(skip(interconnect, rx, cipher))]
325pub(crate) async fn runner(
326    mut interconnect: Interconnect,
327    rx: Receiver<UdpRxMessage>,
328    cipher: Cipher,
329    crypto_mode: CryptoMode,
330    config: Config,
331    udp_socket: UdpSocket,
332    ssrc_signalling: Arc<SsrcTracker>,
333    dave_session: Arc<RwLock<Option<davey::DaveSession>>>,
334    dave_protocol_version: Arc<AtomicU16>,
335) {
336    trace!("UDP receive handle started.");
337
338    let mut state = UdpRx {
339        cipher,
340        crypto_mode,
341        decoder_map: HashMap::new(),
342        config,
343        rx,
344        ssrc_signalling,
345        udp_socket,
346        dave_session,
347        dave_protocol_version,
348    };
349
350    state.run(&mut interconnect).await;
351
352    trace!("UDP receive handle stopped.");
353}
