186    pub(crate) fn do_rebuilds(
187        &mut self,
188        event_failure: bool,
189        conn_failure: bool,
190    ) -> StdResult<(), SendError<CoreMessage>> {
191        // event failure? rebuild interconnect.
192        // ws or udp failure? full connect
193        // (soft reconnect is covered by the ws task.)
194        //
195        // in both cases, send failure is fatal,
196        // but will only occur on disconnect.
197        if event_failure {
198            self.rebuild_interconnect()?;
199        }
200
201        if conn_failure {
202            self.full_reconnect_gateway()?;
203        }
204
205        Ok(())
206    }

220    #[inline]
221    pub(crate) fn handle_message(
222        &mut self,
223        msg: MixerMessage,
224        packet: &mut [u8],
225    ) -> (bool, bool, bool) {
226        let mut events_failure = false;
227        let mut conn_failure = false;
228        let mut should_exit = false;
229
230        let error = match msg {
231            MixerMessage::AddTrack(t) => self.add_track(t),
232            MixerMessage::SetTrack(t) => {
233                self.tracks.clear();
234                self.track_handles.clear();
235
236                let mut out = self.fire_event(EventMessage::RemoveAllTracks);
237
238                if let Some(t) = t {
239                    // Do this unconditionally: this affects local state infallibly,
240                    // with the event installation being the remote part.
241                    if let Err(e) = self.add_track(t) {
242                        out = Err(e);
243                    }
244                }
245
246                out
247            },
248            MixerMessage::SetBitrate(b) => {
249                self.bitrate = b;
250                if let Err(e) = self.set_bitrate(b) {
251                    error!("Failed to update bitrate {:?}", e);
252                }
253                Ok(())
254            },
255            MixerMessage::SetMute(m) => {
256                self.muted = m;
257                Ok(())
258            },
259            MixerMessage::SetConn(conn, ssrc) => {
260                self.conn_active = Some(conn);
261                let mut rtp = MutableRtpPacket::new(packet).expect(
262                    "Too few bytes in self.packet for RTP header.\
263                        (Blame: VOICE_PACKET_MAX?)",
264                );
265                rtp.set_ssrc(ssrc);
266                rtp.set_sequence(random::<u16>().into());
267                rtp.set_timestamp(random::<u32>().into());
268                self.deadline = Instant::now();
269
270                self.update_keepalive(ssrc);
271                Ok(())
272            },
273            MixerMessage::DropConn => {
274                self.conn_active = None;
275                Ok(())
276            },
277            MixerMessage::ReplaceInterconnect(i) => {
278                self.prevent_events = false;
279
280                if let Some(ws) = &self.ws {
281                    conn_failure |= ws.send(WsMessage::ReplaceInterconnect(i.clone())).is_err();
282                }
283
284                #[cfg(feature = "receive")]
285                if let Some(conn) = &self.conn_active {
286                    conn_failure |= conn
287                        .udp_rx
288                        .send(UdpRxMessage::ReplaceInterconnect(i.clone()))
289                        .is_err();
290                }
291
292                self.interconnect = i;
293
294                self.rebuild_tracks()
295            },
296            MixerMessage::SetConfig(new_config) => {
297                if new_config.mix_mode != self.config.mix_mode {
298                    self.soft_clip = SoftClip::new(new_config.mix_mode.to_opus());
299                    if let Ok(enc) = new_encoder(self.bitrate, new_config.mix_mode) {
300                        self.encoder = enc;
301                    } else {
302                        self.bitrate = DEFAULT_BITRATE;
303                        self.encoder = new_encoder(self.bitrate, new_config.mix_mode)
304                            .expect("Failed fallback rebuild of OpusEncoder with safe inputs.");
305                    }
306
307                    let sl = new_config.mix_mode.symph_layout();
308                    self.sample_buffer = SampleBuffer::<f32>::new(
309                        MONO_FRAME_SIZE as u64,
310                        SignalSpec::new_with_layout(SAMPLE_RATE_RAW as u32, sl),
311                    );
312                    self.symph_mix = AudioBuffer::<f32>::new(
313                        MONO_FRAME_SIZE as u64,
314                        SignalSpec::new_with_layout(SAMPLE_RATE_RAW as u32, sl),
315                    );
316                }
317
318                self.config = Arc::new(
319                    #[cfg(feature = "receive")]
320                    new_config.clone(),
321                    #[cfg(not(feature = "receive"))]
322                    new_config,
323                );
324
325                if self.tracks.capacity() < self.config.preallocated_tracks {
326                    self.tracks
327                        .reserve(self.config.preallocated_tracks - self.tracks.len());
328                }
329
330                #[cfg(feature = "receive")]
331                if let Some(conn) = &self.conn_active {
332                    conn_failure |= conn
333                        .udp_rx
334                        .send(UdpRxMessage::SetConfig(new_config))
335                        .is_err();
336                }
337
338                Ok(())
339            },
340            MixerMessage::RebuildEncoder => match new_encoder(self.bitrate, self.config.mix_mode) {
341                Ok(encoder) => {
342                    self.encoder = encoder;
343                    Ok(())
344                },
345                Err(e) => {
346                    error!("Failed to rebuild encoder. Resetting bitrate. {:?}", e);
347                    self.bitrate = DEFAULT_BITRATE;
348                    self.encoder = new_encoder(self.bitrate, self.config.mix_mode)
349                        .expect("Failed fallback rebuild of OpusEncoder with safe inputs.");
350                    Ok(())
351                },
352            },
353            MixerMessage::Ws(new_ws_handle) => {
354                self.ws = new_ws_handle;
355                if let Err(e) = self.send_gateway_speaking() {
356                    conn_failure |= e.should_trigger_connect();
357                }
358                Ok(())
359            },
360            MixerMessage::Poison => {
361                should_exit = true;
362                Ok(())
363            },
364        };
365
366        if let Err(e) = error {
367            events_failure |= e.should_trigger_interconnect_rebuild();
368            conn_failure |= e.should_trigger_connect();
369        }
370
371        (events_failure, conn_failure, should_exit)
372    }

382    pub(crate) fn fire_event(&self, event: EventMessage) -> Result<()> {
383        // As this task is responsible for noticing the potential death of an event context,
384        // it's responsible for not forcibly recreating said context repeatedly.
385        if !self.prevent_events {
386            self.interconnect.events.send(event)?;
387        }
388
389        Ok(())
390    }

407        for (track, handle) in self.tracks.iter().zip(self.track_handles.iter()) {
408            let evts = EventStore::default();
409            let state = track.state();
410            let handle = handle.clone();

415        }
416
417        Ok(())

420    #[inline]
421    pub(crate) fn audio_commands_events(&mut self) -> Result<()> {
422        // Apply user commands.
423        for (i, track) in self.tracks.iter_mut().enumerate() {
424            // This causes fallible event system changes,
425            // but if the event thread has died then we'll certainly
426            // detect that on the tick later.
427            // Changes to play state etc. MUST all be handled.
428            let action = track.process_commands(i, &self.interconnect);
429
430            if let Some(req) = action.seek_point {
431                track.seek(
432                    i,
433                    req,
434                    &self.interconnect,
435                    &self.thread_pool,
436                    &self.config,
437                    self.prevent_events,
438                );
439            }
440
441            if let Some(callback) = action.make_playable {
442                if let Err(e) = track.get_or_ready_input(
443                    i,
444                    &self.interconnect,
445                    &self.thread_pool,
446                    &self.config,
447                    self.prevent_events,
448                ) {
449                    track.callbacks.make_playable = Some(callback);
450                    if let Some(fail) = e.as_user() {
451                        track.playing = PlayMode::Errored(fail);
452                    }
453                    if let Some(req) = e.into_seek_request() {
454                        track.seek(
455                            i,
456                            req,
457                            &self.interconnect,
458                            &self.thread_pool,
459                            &self.config,
460                            self.prevent_events,
461                        );
462                    }
463                } else {
464                    // Track is already ready: don't register callback and just act.
465                    drop(callback.send(Ok(())));
466                }
467            }
468        }
469
470        let mut i = 0;
471        while i < self.tracks.len() {
472            let track = self
473                .tracks
474                .get_mut(i)
475                .expect("Tried to remove an illegal track index.");
476
477            if track.playing.is_done() {
478                let p_state = track.playing.clone();
479                let to_drop = self.tracks.swap_remove(i);
480                self.disposer
481                    .dispose(DisposalMessage::Track(Box::new(to_drop)));
482
483                let to_drop = self.track_handles.swap_remove(i);
484                self.disposer.dispose(DisposalMessage::Handle(to_drop));
485
486                self.fire_event(EventMessage::ChangeState(
487                    i,
488                    TrackStateChange::Mode(p_state),
489                ))?;
490            } else {
491                i += 1;
492            }
493        }
494
495        // Tick -- receive side also handles removals in same manner after it increments
496        // times etc.
497        self.fire_event(EventMessage::Tick)?;
498
499        Ok(())
500    }

523    #[inline]
524    pub fn mix_and_build_packet(&mut self, packet: &mut [u8]) -> Result<usize> {
525        // symph_mix is an `AudioBuffer` (planar format), we need to convert this
526        // later into an interleaved `SampleBuffer` for libopus.
527        self.symph_mix.clear();
528        self.symph_mix.render_reserved(Some(MONO_FRAME_SIZE));
529        self.resample_scratch.clear();
530
531        // Walk over all the audio files, combining into one audio frame according
532        // to volume, play state, etc.
533        let mut mix_len = {
534            let out = self.mix_tracks(packet);
535
536            self.sample_buffer.copy_interleaved_typed(&self.symph_mix);
537
538            out
539        };
540
541        if self.muted {
542            mix_len = MixType::MixedPcm(0);
543        }
544
545        // Explicit "Silence" frame handling: if there is no mixed data, we must send
546        // ~5 frames of silence (unless another good audio frame appears) before we
547        // stop sending RTP frames.
548        if mix_len == MixType::MixedPcm(0) {
549            if self.silence_frames > 0 {
550                self.silence_frames -= 1;
551                let mut rtp = MutableRtpPacket::new(packet).expect(
552                    "FATAL: Too few bytes in self.packet for RTP header.\
553                        (Blame: VOICE_PACKET_MAX?)",
554                );
555
556                let payload = rtp.payload_mut();
557                let pre_len = self.crypto_mode().payload_prefix_len();
558
559                payload[pre_len..pre_len + SILENT_FRAME.len()].copy_from_slice(&SILENT_FRAME[..]);
560
561                mix_len = MixType::Passthrough(SILENT_FRAME.len());
562            } else {
563                // Per official guidelines, send 5x silence BEFORE we stop speaking.
564                return Ok(0);
565            }
566        } else {
567            self.silence_frames = 5;
568
569            if let MixType::MixedPcm(n) = mix_len {
570                if self.config.use_softclip {
571                    self.soft_clip.apply(
572                        (&mut self.sample_buffer.samples_mut()
573                            [..n * self.config.mix_mode.channels()])
574                            .try_into()
575                            .expect("Mix buffer is known to have a valid sample count (softclip)."),
576                    )?;
577                }
578            }
579        }
580
581        // For the benefit of test cases, send the raw un-RTP'd data.
582        #[cfg(test)]
583        let out = if let Some(OutputMode::Raw(_)) = &self.config.override_connection {
584            // This case has been handled before buffer clearing above.
585            let msg = match mix_len {
586                MixType::Passthrough(len) if len == SILENT_FRAME.len() => OutputMessage::Silent,
587                MixType::Passthrough(len) => {
588                    let rtp = RtpPacket::new(packet).expect(
589                        "FATAL: Too few bytes in self.packet for RTP header.\
590                            (Blame: VOICE_PACKET_MAX?)",
591                    );
592                    let payload = rtp.payload();
593                    let opus_frame =
594                        (payload[self.crypto_mode().payload_prefix_len()..][..len]).to_vec();
595
596                    OutputMessage::Passthrough(opus_frame)
597                },
598                MixType::MixedPcm(_) => OutputMessage::Mixed(
599                    self.sample_buffer.samples()[..self.config.mix_mode.sample_count_in_frame()]
600                        .to_vec(),
601                ),
602            };
603
604            self.raw_msg = Some(msg);
605
606            Ok(1)
607        } else {
608            self.prep_packet(mix_len, packet)
609        };
610
611        #[cfg(not(test))]
612        let out = self.prep_packet(mix_len, packet);
613
614        // Zero out all planes of the mix buffer if any audio was written.
615        if matches!(mix_len, MixType::MixedPcm(a) if a > 0) {
616            for plane in self.symph_mix.planes_mut().planes() {
617                plane.fill(0.0);
618            }
619        }
620
621        out
622    }

624    #[inline]
625    fn prep_packet(&mut self, mix_len: MixType, packet: &mut [u8]) -> Result<usize> {
626        let send_buffer = self.sample_buffer.samples();
627
628        let conn = self
629            .conn_active
630            .as_mut()
631            .expect("Shouldn't be mixing packets without access to a cipher + UDP dest.");
632
633        let mut rtp = MutableRtpPacket::new(packet).expect(
634            "FATAL: Too few bytes in self.packet for RTP header.\
635                (Blame: VOICE_PACKET_MAX?)",
636        );
637
638        let payload = rtp.payload_mut();
639        let crypto_mode = conn.crypto_state.kind();
640        let first_payload_byte = crypto_mode.payload_prefix_len();
641
642        // If passthrough, Opus payload in place already.
643        // Else encode into buffer with space for AEAD encryption headers.
644        let payload_len = match mix_len {
645            MixType::Passthrough(opus_len) => opus_len,
646            MixType::MixedPcm(_samples) => {
647                let total_payload_space = payload.len() - crypto_mode.payload_suffix_len();
648                self.encoder.encode_float(
649                    &send_buffer[..self.config.mix_mode.sample_count_in_frame()],
650                    &mut payload[first_payload_byte..total_payload_space],
651                )?
652            },
653        };
654
655        let final_payload_size = conn
656            .crypto_state
657            .write_packet_nonce(&mut rtp, first_payload_byte + payload_len);
658
659        // Packet encryption ignored in test modes.
660        #[cfg(not(test))]
661        let encrypt = true;
662        #[cfg(test)]
663        let encrypt = self.config.override_connection.is_none();
664
665        if encrypt {
666            conn.cipher
667                .encrypt_pkt_in_place(&mut rtp, final_payload_size)?;
668        }
669
670        Ok(RtpPacket::minimum_packet_size() + final_payload_size)
671    }

673    #[inline]
674    pub(crate) fn send_packet(&self, packet: &[u8]) -> Result<()> {
675        #[cfg(test)]
676        let send_status = if let Some(OutputMode::Raw(tx)) = &self.config.override_connection {
677            // This case has been handled before buffer clearing in `mix_and_build_packet`.
678            drop(tx.send(self.raw_msg.clone().unwrap().into()));
679
680            Ok(())
681        } else {
682            self.send_packet_(packet)
683        };
684
685        #[cfg(not(test))]
686        let send_status = self.send_packet_(packet);
687
688        send_status.or_else(Error::disarm_would_block)?;
689
690        Ok(())
691    }

693    #[inline]
694    fn send_packet_(&self, packet: &[u8]) -> Result<()> {
695        let conn = self
696            .conn_active
697            .as_ref()
698            .expect("Shouldn't be mixing packets without access to a cipher + UDP dest.");
699
700        #[cfg(test)]
701        if let Some(OutputMode::Rtp(tx)) = &self.config.override_connection {
702            // Test mode: send unencrypted (compressed) packets to local receiver.
703            drop(tx.send(packet.to_vec().into()));
704        } else {
705            conn.udp_tx.send(packet)?;
706        }
707
708        #[cfg(not(test))]
709        {
710            // Normal operation: send encrypted payload to UDP Tx task.
711            conn.udp_tx.send(packet)?;
712        }
713
714        Ok(())
715    }

717    #[inline]
718    pub(crate) fn check_and_send_keepalive(&mut self, now: Option<Instant>) -> Result<()> {
719        if let Some(conn) = self.conn_active.as_mut() {
720            let now = now.unwrap_or_else(Instant::now);
721            if now >= self.keepalive_deadline {
722                conn.udp_tx.send(&self.keepalive_packet)?;
723                self.keepalive_deadline += UDP_KEEPALIVE_GAP;
724            }
725        }
726
727        Ok(())
728    }

732        if let Some(ws) = &self.ws {
733            ws.send(WsMessage::Speaking(true))?;
734        }

736        Ok(())
737    }

739    #[inline]
740    pub(crate) fn send_gateway_not_speaking(&self) {
741        if let Some(ws) = &self.ws {
742            // NOTE: this explicit `drop` should prevent a catastrophic thread pileup.
743            // A full reconnect might cause an inner closed connection.
744            // It's safer to leave the central task to clean this up and
745            // pass the mixer a new channel.
746            drop(ws.send(WsMessage::Speaking(false)));
747        }
748    }

750    #[inline]
751    fn mix_tracks(&mut self, packet: &mut [u8]) -> MixType {
752        // Get a slice of bytes to write in data for Opus packet passthrough.
753        let mut rtp = MutableRtpPacket::new(packet).expect(
754            "FATAL: Too few bytes in self.packet for RTP header.\
755                (Blame: VOICE_PACKET_MAX?)",
756        );
757        let payload = rtp.payload_mut();
758        let opus_frame = &mut payload[self.crypto_mode().payload_prefix_len()..];
759
760        // Opus frame passthrough.
761        // This requires that we have only one PLAYING track, who has volume 1.0, and an
762        // Opus codec type (verified later in mix_symph_indiv).
763        //
764        // We *could* cache the number of live tracks separately, but that makes this
765        // quite fragile given all the ways a user can alter the PlayMode.
766        let mut num_live = 0;
767        let mut last_live_vol = 1.0;
768        for track in &self.tracks {
769            if track.playing.is_playing() {
770                num_live += 1;
771                last_live_vol = track.volume;
772            }
773        }
774        let do_passthrough = num_live == 1 && (last_live_vol - 1.0).abs() < f32::EPSILON;
775
776        let mut len = 0;
777        for (i, track) in self.tracks.iter_mut().enumerate() {
778            let vol = track.volume;
779
780            // This specifically tries to get tracks who are "preparing",
781            // so that event handlers and the like can all be fired without
782            // the track being in a `Play` state.
783            if !track.should_check_input() {
784                continue;
785            }
786
787            let should_play = track.playing.is_playing();
788
789            let input = track.get_or_ready_input(
790                i,
791                &self.interconnect,
792                &self.thread_pool,
793                &self.config,
794                self.prevent_events,
795            );
796
797            let (input, mix_state) = match input {
798                Ok(i) => i,
799                Err(InputReadyingError::Waiting) => continue,
800                Err(InputReadyingError::NeedsSeek(req)) => {
801                    track.seek(
802                        i,
803                        req,
804                        &self.interconnect,
805                        &self.thread_pool,
806                        &self.config,
807                        self.prevent_events,
808                    );
809                    continue;
810                },
811                // TODO: allow for retry in given time.
812                Err(e) => {
813                    if let Some(fail) = e.as_user() {
814                        track.playing = PlayMode::Errored(fail);
815                    }
816                    continue;
817                },
818            };
819
820            // Now that we have dealt with potential errors in preparing tracks,
821            // only do any mixing if the track is to be played!
822            if !should_play {
823                continue;
824            }
825
826            let (mix_type, status) = mix_logic::mix_symph_indiv(
827                &mut self.symph_mix,
828                &mut self.resample_scratch,
829                input,
830                mix_state,
831                vol,
832                do_passthrough.then_some(&mut *opus_frame),
833            );
834
835            let return_here = if let MixType::MixedPcm(pcm_len) = mix_type {
836                len = len.max(pcm_len);
837                false
838            } else {
839                if mix_state.passthrough == Passthrough::Inactive {
840                    input.decoder.reset();
841                }
842                mix_state.passthrough = Passthrough::Active;
843                true
844            };
845
846            // FIXME: allow Ended to trigger a seek/loop/revisit in the same mix cycle?
847            // Would this be possible with special-casing to mark some inputs as fast
848            // to recreate? Probably not doable in the general case.
849            match status {
850                MixStatus::Live => track.step_frame(),
851                MixStatus::Errored(e) =>
852                    track.playing = PlayMode::Errored(PlayError::Decode(e.into())),
853                MixStatus::Ended if track.do_loop() => {
854                    drop(self.track_handles[i].seek(Duration::default()));
855                    if !self.prevent_events {
856                        // position update is sent out later, when the seek concludes.
857                        drop(self.interconnect.events.send(EventMessage::ChangeState(
858                            i,
859                            TrackStateChange::Loops(track.loops, false),
860                        )));
861                    }
862                },
863                MixStatus::Ended => {
864                    track.end();
865                },
866            }
867
868            // This needs to happen here due to borrow checker shenanigans.
869            if return_here {
870                return mix_type;
871            }
872        }
873
874        MixType::MixedPcm(len)
875    }
