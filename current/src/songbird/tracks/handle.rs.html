8#[derive(Clone, Debug)]
9/// Handle for safe control of a [`Track`] from other threads, outside
10/// of the audio mixing and voice handling context.
11///
12/// These are cheap to clone, using `Arc<...>` internally.
13///
14/// Many method calls here are fallible; in most cases, this will be because
15/// the underlying [`Track`] object has been discarded. Those which aren't refer
16/// to shared data not used by the driver.
17pub struct TrackHandle {
18    inner: Arc<InnerHandle>,
19}
20
21#[derive(Debug)]
22struct InnerHandle {
23    command_channel: Sender<TrackCommand>,
24    uuid: Uuid,
25    data: Arc<dyn Any + Send + Sync + 'static>,
26}
27
28impl TrackHandle {
29    /// Creates a new handle, using the given command sink.
30    ///
31    /// [`Input`]: crate::input::Input
32    #[must_use]
33    pub(crate) fn new(
34        command_channel: Sender<TrackCommand>,
35        uuid: Uuid,
36        data: Arc<dyn Any + Send + Sync + 'static>,
37    ) -> Self {
38        let inner = Arc::new(InnerHandle {
39            command_channel,
40            uuid,
41            data,
42        });
43
44        Self { inner }
45    }

72    #[must_use]
73    /// Ready a track for playing if it is lazily initialised.
74    ///
75    /// If a track is already playable, the callback will instantly succeed.
76    pub fn make_playable(&self) -> TrackCallback<()> {
77        let (tx, rx) = flume::bounded(1);
78        let fail = self.send(TrackCommand::MakePlayable(tx)).is_err();
79
80        TrackCallback { fail, rx }
81    }
82
83    /// Ready a track for playing if it is lazily initialised.
84    ///
85    /// This folds [`Self::make_playable`] into a single `async` result, but must
86    /// be awaited for the command to be sent.
87    pub async fn make_playable_async(&self) -> TrackResult<()> {
88        self.make_playable().result_async().await
89    }

91    #[must_use]
92    /// Seeks along the track to the specified position.
93    ///
94    /// If the underlying [`Input`] does not support seeking,
95    /// forward seeks will succeed. Backward seeks will recreate the
96    /// track using the lazy [`Compose`] if present. The returned callback
97    /// will indicate whether the seek succeeded.
98    ///
99    /// [`Input`]: crate::input::Input
100    /// [`Compose`]: crate::input::Compose
101    pub fn seek(&self, position: Duration) -> TrackCallback<Duration> {
102        let (tx, rx) = flume::bounded(1);
103        let fail = self
104            .send(TrackCommand::Seek(SeekRequest {
105                time: position,
106                callback: tx,
107            }))
108            .is_err();
109
110        TrackCallback { fail, rx }
111    }
112
113    /// Seeks along the track to the specified position.
114    ///
115    /// This folds [`Self::seek`] into a single `async` result, but must
116    /// be awaited for the command to be sent.
117    pub async fn seek_async(&self, position: Duration) -> TrackResult<Duration> {
118        self.seek(position).result_async().await
119    }

121    /// Attach an event handler to an audio track. These will receive [`EventContext::Track`].
122    ///
123    /// Events which can only be fired by the global context return [`ControlError::InvalidTrackEvent`]
124    ///
125    /// [`EventContext::Track`]: crate::events::EventContext::Track
126    pub fn add_event<F: EventHandler + 'static>(&self, event: Event, action: F) -> TrackResult<()> {
127        let cmd = TrackCommand::AddEvent(EventData::new(event, action));
128        if event.is_global_only() {
129            Err(ControlError::InvalidTrackEvent)
130        } else {
131            self.send(cmd)
132        }
133    }
134
135    /// Perform an arbitrary synchronous action on a raw [`Track`] object.
136    ///
137    /// This will give access to a [`View`] of the current track state and [`Metadata`],
138    /// which can be used to take an [`Action`].
139    ///
140    /// Users **must** ensure that no costly work or blocking occurs
141    /// within the supplied function or closure. *Taking excess time could prevent
142    /// timely sending of packets, causing audio glitches and delays*.
143    ///
144    /// [`Metadata`]: crate::input::Metadata
145    pub fn action<F>(&self, action: F) -> TrackResult<()>
146    where
147        F: FnOnce(View<'_>) -> Option<Action> + Send + Sync + 'static,
148    {
149        self.send(TrackCommand::Do(Box::new(action)))
150    }
151
152    /// Request playback information and state from the audio context.
153    pub async fn get_info(&self) -> TrackResult<TrackState> {
154        let (tx, rx) = flume::bounded(1);
155        self.send(TrackCommand::Request(tx))?;
156
157        rx.recv_async().await.map_err(|_| ControlError::Finished)
158    }

160    /// Set an audio track to loop indefinitely.
161    ///
162    /// This requires either a [`Compose`] to be present or for the
163    /// input stream to be seekable.
164    ///
165    /// [`Input`]: crate::input::Input
166    /// [`Compose`]: crate::input::Compose

168        self.send(TrackCommand::Loop(LoopState::Infinite))
169    }

171    /// Set an audio track to no longer loop.
172    ///
173    /// This follows the same rules as [`enable_loop`].

175    /// [`enable_loop`]: Self::enable_loop

177        self.send(TrackCommand::Loop(LoopState::Finite(NonMaxU32::ZERO)))

195    /// Allows access to this track's attached Data.
196    ///
197    /// Data allows additional, user-defined data shared by all handles
198    /// to be attached to any track.
199    ///
200    /// # Panics
201    /// This method will panic if the Data has not been initialised, or the type
202    /// provided does not equal the type passed to [`Track::new_with_data`].
203    #[must_use]
204    pub fn data<Data>(&self) -> Arc<Data>
205    where
206        Data: Send + Sync + 'static,
207    {
208        Arc::clone(&self.inner.data)
209            .downcast()
210            .expect("TrackHandle::data generic does not match type set in TrackHandle::set_data")
211    }

213    #[inline]
214    /// Send a raw command to the [`Track`] object.
215    ///
216    /// [`Track`]: Track
217    pub(crate) fn send(&self, cmd: TrackCommand) -> TrackResult<()> {
218        // As the send channels are unbounded, we can be reasonably certain
219        // that send failure == cancellation.
220        self.inner
221            .command_channel
222            .send(cmd)
223            .map_err(|_e| ControlError::Finished)
224    }
225}

227/// Asynchronous reply for an operation applied to a [`TrackHandle`].
228///
229/// This object does not need to be `.await`ed for the driver to perform an action.
230/// Async threads can then call, e.g., [`TrackHandle::make_playable`], and safely drop
231/// this callback if the result isn't needed.
232pub struct TrackCallback<T> {
233    fail: bool,
234    rx: Receiver<Result<T, PlayError>>,
235}
236
237impl<T> TrackCallback<T> {
238    /// Consumes this handle to await a reply from the driver, blocking the current thread.
239    pub fn result(self) -> TrackResult<T> {
240        if self.fail {
241            Err(ControlError::Finished)
242        } else {
243            self.rx.recv()?.map_err(ControlError::Play)
244        }
245    }
246
247    /// Consumes this handle to await a reply from the driver asynchronously.
248    pub async fn result_async(self) -> TrackResult<T> {
249        if self.fail {
250            Err(ControlError::Finished)
251        } else {
252            self.rx.recv_async().await?.map_err(ControlError::Play)
253        }
254    }
255
256    #[must_use]
257    /// Returns `true` if the operation instantly failed due to the target track being
258    /// removed.
259    pub fn is_hung_up(&self) -> bool {
260        self.fail
261    }
262}

283    #[tokio::test]
284    #[ntest::timeout(10_000)]
285    async fn seek_callback_fires() {
286        let (t_handle, config) = Config::test_cfg(true);
287        let mut driver = Driver::new(config.clone());
288
289        let file = File::new(FILE_WAV_TARGET);
290        let handle = driver.play(Track::from(file).pause());
291
292        let target = Duration::from_millis(500);
293        let callback = handle.seek(target);
294        t_handle.spawn_ticker();
295
296        let answer = callback.result_async().await;
297        assert!(answer.is_ok());
298        let answer = answer.unwrap();
299        let delta = Duration::from_millis(100);
300        assert!(answer > target - delta && answer < target + delta);
301    }
