17/// The send-half of a worker thread, with bookkeeping mechanisms to help
18/// the idle task schedule incoming tasks.
19pub struct Worker {
20    id: WorkerId,
21    stats: Arc<LiveStatBlock>,
22    config: Config,
23    tx: Sender<(TaskId, ParkedMixer)>,
24    known_empty_since: Option<TokInstant>,
25}

31        config: Config,
32        sched_tx: Sender<SchedulerMessage>,
33        global_stats: Arc<StatBlock>,

57    /// Mark the worker thread as idle from the present time if it reports no tasks.
58    ///
59    /// This time information is used for thread culling.
60    #[inline]
61    pub fn try_mark_empty(&mut self, now: TokInstant) -> Option<TokInstant> {
62        if self.stats.live_mixers() == 0 {
63            self.known_empty_since.get_or_insert(now);
64        } else {
65            self.mark_busy();
66        }
67
68        self.known_empty_since
69    }

72    #[inline]
73    pub fn mark_busy(&mut self) {
74        self.known_empty_since = None;
75    }

80        self.known_empty_since.is_none()
81    }

191    /// Returns whether the loop should exit (i.e., culled by main `Scheduler`).
192    #[inline]
193    pub fn run_once(&mut self) -> bool {
194        // Check for new tasks.
195        if self.handle_scheduler_msgs().is_err() {
196            return false;
197        }
198
199        // Receive commands for each task.
200        self.handle_task_msgs();
201
202        // Move any idle calls back to the global pool.
203        self.demote_and_remove_mixers();
204
205        // Take a clock measure before and after each packet.
206        let mut pre_pkt_time = Instant::now();
207        let mut worst_task = (0, Duration::default());
208
209        for (i, (packet_len, mixer)) in self
210            .packet_lens
211            .iter_mut()
212            .zip(self.tasks.iter_mut())
213            .enumerate()
214        {
215            let (block, inner) = get_memory_indices(i);
216            match mixer.mix_and_build_packet(&mut self.packets[block][inner..][..VOICE_PACKET_MAX])
217            {
218                Ok(written_sz) => *packet_len = written_sz,
219                e => {
220                    *packet_len = 0;
221                    rebuild_if_err(mixer, e, &mut self.to_cull, i);
222                },
223            }
224            let post_pkt_time = Instant::now();
225            let cost = post_pkt_time.duration_since(pre_pkt_time);
226            if cost > worst_task.1 {
227                worst_task = (i, cost);
228            }
229            pre_pkt_time = post_pkt_time;
230        }
231
232        let end_of_work = pre_pkt_time;
233
234        if let Some(start_of_work) = self.start_of_work {
235            let ns_cost = self.stats.store_compute_cost(end_of_work - start_of_work);
236
237            if self.config.move_expensive_tasks
238                && ns_cost >= RESCHEDULE_THRESHOLD
239                && self.ids.len() > 1
240            {
241                self.offload_mixer(worst_task.0, worst_task.1);
242            }
243        }
244
245        self.timed_remove_excess_blocks(end_of_work);
246
247        // Wait till the right time to send this packet:
248        // usually a 20ms tick, in test modes this is either a finite number of runs or user input.
249        self.march_deadline();
250
251        // Send all.
252        self.start_of_work = Some(Instant::now());
253        for (i, (packet_len, mixer)) in self
254            .packet_lens
255            .iter_mut()
256            .zip(self.tasks.iter_mut())
257            .enumerate()
258        {
259            let (block, inner) = get_memory_indices(i);
260            let packet = &mut self.packets[block][inner..];
261            if *packet_len > 0 {
262                let res = mixer.send_packet(&packet[..*packet_len]);
263                rebuild_if_err(mixer, res, &mut self.to_cull, i);
264            }
265            #[cfg(test)]
266            if *packet_len == 0 {
267                mixer.test_signal_empty_tick();
268            }
269            advance_rtp_counters(packet);
270        }
271
272        for (i, mixer) in self.tasks.iter_mut().enumerate() {
273            let res = mixer
274                .audio_commands_events()
275                .and_then(|()| mixer.check_and_send_keepalive(self.start_of_work));
276            rebuild_if_err(mixer, res, &mut self.to_cull, i);
277        }
278
279        true
280    }

282    #[cfg(test)]
283    fn _march_deadline(&mut self) {
284        // For testing, assume all will have same tick style.
285        // Only count 'remaining loops' on one of the nodes.
286        let mixer = self.tasks.get_mut(0).map(|m| {
287            let style = m.config.tick_style.clone();
288            (m, style)
289        });
290
291        match mixer {
292            None | Some((_, TickStyle::Timed)) => {
293                std::thread::sleep(self.deadline.saturating_duration_since(Instant::now()));
294                self.deadline += TIMESTEP_LENGTH;
295            },
296            Some((m, TickStyle::UntimedWithExecLimit(rx))) => {
297                if m.remaining_loops.is_none() {
298                    if let Ok(new_val) = rx.recv() {
299                        m.remaining_loops = Some(new_val.wrapping_sub(1));
300                    }
301                }
302
303                if let Some(cnt) = m.remaining_loops.as_mut() {
304                    if *cnt == 0 {
305                        m.remaining_loops = None;
306                    } else {
307                        *cnt = cnt.wrapping_sub(1);
308                    }
309                }
310            },
311        }
312    }

333    #[inline]
334    fn handle_scheduler_msgs(&mut self) -> Result<(), ()> {
335        let mut activation_time = None;
336        loop {
337            match self.rx.try_recv() {
338                Ok((id, task)) => {
339                    self.add_task(
340                        task,
341                        id,
342                        *activation_time.get_or_insert_with(|| {
343                            self.deadline
344                                .checked_sub(TIMESTEP_LENGTH)
345                                .unwrap_or(self.deadline)
346                        }),
347                    );
348                },
349                Err(TryRecvError::Empty) => break,
350                Err(TryRecvError::Disconnected) => return Err(()),
351            }
352        }
353
354        Ok(())
355    }

357    /// Handle messages from each tasks's `Driver`, marking dead tasks for removal.
358    #[inline]
359    fn handle_task_msgs(&mut self) {
360        for (i, (packet, mixer)) in self
361            .packets
362            .iter_mut()
363            .flat_map(|v| v.chunks_exact_mut(VOICE_PACKET_MAX))
364            .zip(self.tasks.iter_mut())
365            .enumerate()
366        {
367            let mut events_failure = false;
368            let mut conn_failure = false;
369
370            let fatal = loop {
371                match mixer.mix_rx.try_recv() {
372                    Ok(m) => {
373                        let (events, conn, should_exit) = mixer.handle_message(m, packet);
374                        events_failure |= events;
375                        conn_failure |= conn;
376
377                        if should_exit {
378                            break true;
379                        }
380                    },
381                    Err(TryRecvError::Disconnected) => {
382                        break true;
383                    },
384
385                    Err(TryRecvError::Empty) => {
386                        break false;
387                    },
388                }
389            };
390
391            if fatal || mixer.do_rebuilds(events_failure, conn_failure).is_err() {
392                // this is not zipped in because it is *not* needed most ticks.
393                self.to_cull[i] = true;
394            }
395        }
396    }

404    /// Check and demote for any tasks without live audio sources who have sent all
405    /// necessary silent frames (or remove dead tasks).
406    ///
407    /// This must occur *after* handling per-track events to prevent erroneously
408    /// descheduling tasks.
409    #[inline]
410    pub fn demote_and_remove_mixers(&mut self) {
411        let mut i = 0;
412        while i < self.tasks.len() {
413            #[cfg(test)]
414            let force_conn = self.tasks[i].config.override_connection.is_some();
415            #[cfg(not(test))]
416            let force_conn = false;
417
418            // Benchmarking suggests that these asserts remove some bounds checks for us.
419            assert!(i < self.tasks.len());
420            assert!(i < self.to_cull.len());
421
422            if self.to_cull[i]
423                || (self.tasks[i].tracks.is_empty() && self.tasks[i].silence_frames == 0)
424                || !(self.tasks[i].conn_active.is_some() || force_conn)
425            {
426                self.stats.remove_mixer();
427
428                if let Some((id, parked)) = self.remove_task(i) {
429                    self.global_stats.move_mixer_to_idle();
430                    _ = self.tx.send(SchedulerMessage::Demote(id, parked));
431                } else {
432                    self.global_stats.remove_live_mixer();
433                }
434            } else {
435                i += 1;
436            }
437        }
438    }

440    /// Return a given mixer to the main scheduler if this worker is overloaded.
441    #[inline]
442    pub fn offload_mixer(&mut self, idx: usize, cost: Duration) {
443        self.stats.remove_mixer();
444
445        if let Some((id, mut parked)) = self.remove_task(idx) {
446            self.global_stats.move_mixer_to_idle();
447            parked.last_cost = Some(cost);
448            _ = self
449                .tx
450                .send(SchedulerMessage::Overspill(self.id, id, parked));
451        } else {
452            self.global_stats.remove_live_mixer();
453        }
454    }

458        let div = self.ids.len() / PACKETS_PER_BLOCK;
459        let rem = self.ids.len() % PACKETS_PER_BLOCK;
460        (rem != 0) as usize + div

473    /// Try to offload excess packet buffers.
474    ///
475    /// If there is currently overallocation, then store the first time at which
476    /// this was seenb. If this condition persists past `MEMORY_CULL_TIMER`, remove
477    /// unnecessary blocks.
478    #[inline]
479    fn timed_remove_excess_blocks(&mut self, now: Instant) {
480        if self.has_excess_blocks() {
481            if let Some(mark_time) = self.excess_buffer_cull_time {
482                if now.duration_since(mark_time) >= MEMORY_CULL_TIMER {
483                    self.remove_excess_blocks();
484                    self.excess_buffer_cull_time = None;
485                }
486            } else {
487                self.excess_buffer_cull_time = Some(now);
488            }
489        } else {
490            self.excess_buffer_cull_time = None;
491        }
492    }

494    #[inline]
495    fn add_task(&mut self, task: ParkedMixer, id: TaskId, activation_time: Instant) {
496        let idx = self.ids.len();
497
498        let elapsed = task.park_time - activation_time;
499
500        let samples_f64 = elapsed.as_secs_f64() * (SAMPLE_RATE_RAW as f64);
501        let mod_samples = (samples_f64 as u64) as u32;
502        let rtp_timestamp = task.rtp_timestamp.wrapping_add(mod_samples);
503
504        self.ids.push(id);
505        self.tasks.push(task.mixer);
506        self.packet_lens.push(0);
507        self.to_cull.push(false);
508
509        let (block, inner_idx) = get_memory_indices(idx);
510
511        while self.packets.len() <= block {
512            self.add_packet_block();
513        }
514        let packet = &mut self.packets[block][inner_idx..][..VOICE_PACKET_MAX];
515
516        let mut rtp = MutableRtpPacket::new(packet).expect(
517            "FATAL: Too few bytes in self.packet for RTP header.\
518                (Blame: VOICE_PACKET_MAX?)",
519        );
520        rtp.set_ssrc(task.ssrc);
521        rtp.set_timestamp(rtp_timestamp.into());
522        rtp.set_sequence(task.rtp_sequence.into());
523    }

525    /// Allocate and store a new packet block.
526    ///
527    /// This will be full-size (`PACKETS_PER_BLOCK`) unless this block
528    /// is a) the last required for the task limit and b) this limit
529    /// is not aligned to `PACKETS_PER_BLOCK`.
530    #[inline]
531    fn add_packet_block(&mut self) {
532        let n_packets = if let Some(limit) = self.config.strategy.task_limit() {
533            let (block, inner) = get_memory_indices_unscaled(limit);
534            if self.packets.len() < block || inner == 0 {
535                PACKETS_PER_BLOCK
536            } else {
537                inner
538            }
539        } else {
540            PACKETS_PER_BLOCK
541        };
542        self.packets.push(packet_block(n_packets));
543    }

564    /// Remove a `Mixer`, returning it to the idle scheduler.
565    ///
566    /// This operates by `swap_remove`ing each element of a Mixer's state, including
567    /// on RTP packet headers. This is achieved by setting up a memcpy between
568    /// buffer segments.
569    #[inline]
570    pub fn remove_task(&mut self, idx: usize) -> Option<(TaskId, ParkedMixer)> {
571        let end = self.tasks.len() - 1;
572
573        let id = self.ids.swap_remove(idx);
574        let _len = self.packet_lens.swap_remove(idx);
575        let mixer = self.tasks.swap_remove(idx);
576        let alive = !self.to_cull.swap_remove(idx);
577
578        let (block, inner_idx) = get_memory_indices(idx);
579
580        let (removed, replacement) = if end > idx {
581            let (end_block, end_inner) = get_memory_indices(end);
582            let (rest, target_block) = self.packets.split_at_mut(end_block);
583            let (last_block, end_pkt) = target_block[0].split_at_mut(end_inner);
584
585            if end_block == block {
586                (&mut last_block[inner_idx..], Some(end_pkt))
587            } else {
588                (&mut rest[block][inner_idx..], Some(end_pkt))
589            }
590        } else {
591            (&mut self.packets[block][inner_idx..], None)
592        };
593
594        let rtp = RtpPacket::new(removed).expect(
595            "FATAL: Too few bytes in self.packet for RTP header.\
596                (Blame: VOICE_PACKET_MAX?)",
597        );
598        let ssrc = rtp.get_ssrc();
599        let rtp_timestamp = rtp.get_timestamp().into();
600        let rtp_sequence = rtp.get_sequence().into();
601
602        if let Some(replacement) = replacement {
603            // Copy the whole packet header since we know it'll be 4B aligned.
604            // 'Just necessary fields' is 2B aligned.
605            const COPY_LEN: usize = RtpPacket::minimum_packet_size();
606            removed[..COPY_LEN].copy_from_slice(&replacement[..COPY_LEN]);
607        }
608
609        alive.then(move || {
610            let park_time = Instant::now();
611
612            (
613                id,
614                ParkedMixer {
615                    mixer,
616                    ssrc,
617                    rtp_sequence,
618                    rtp_timestamp,
619                    park_time,
620                    last_cost: None,
621                    cull_handle: None,
622                },
623            )
624        })
625    }

635/// Initialises a packet block of the required size, prefilling any constant RTP data.
636#[inline]
637fn packet_block(n_packets: usize) -> Box<[u8]> {
638    let mut packets = vec![0u8; VOICE_PACKET_MAX * n_packets].into_boxed_slice();
639
640    for packet in packets.chunks_exact_mut(VOICE_PACKET_MAX) {
641        let mut rtp = MutableRtpPacket::new(packet).expect(
642            "FATAL: Too few bytes in self.packet for RTP header.\
643                (Blame: VOICE_PACKET_MAX?)",
644        );
645        rtp.set_version(RTP_VERSION);
646        rtp.set_payload_type(RTP_PROFILE_TYPE);
647    }
648
649    packets
650}

652/// Returns the block index into `self.packets` and the packet number in
653/// the block for a given worker's index.
654#[inline]
655fn get_memory_indices_unscaled(idx: usize) -> (usize, usize) {
656    let block_size = PACKETS_PER_BLOCK;
657    (idx / block_size, idx % block_size)
658}
659
660/// Returns the block index into `self.packets` and the byte offset into
661/// a packet block for a given worker's index.
662#[inline]
663fn get_memory_indices(idx: usize) -> (usize, usize) {
664    let (block, inner_unscaled) = get_memory_indices_unscaled(idx);
665    (block, inner_unscaled * VOICE_PACKET_MAX)
666}

668#[inline]
669fn advance_rtp_counters(packet: &mut [u8]) {
670    let mut rtp = MutableRtpPacket::new(packet).expect(
671        "FATAL: Too few bytes in self.packet for RTP header.\
672            (Blame: VOICE_PACKET_MAX?)",
673    );
674    rtp.set_sequence(rtp.get_sequence() + 1);
675    rtp.set_timestamp(rtp.get_timestamp() + MONO_FRAME_SIZE as u32);
676}

678/// Structured slightly confusingly: we only want to even access `cull_markers`
679/// in the event of error.
680#[inline]
681fn rebuild_if_err<T>(
682    mixer: &mut Box<Mixer>,
683    res: Result<T, DriverError>,
684    cull_markers: &mut [bool],
685    idx: usize,
686) {
687    if let Err(e) = res {
688        cull_markers[idx] |= mixer
689            .do_rebuilds(
690                e.should_trigger_interconnect_rebuild(),
691                e.should_trigger_connect(),
692            )
693            .is_err();
694    }

704        let rtp = RtpPacket::new(pkt).unwrap();

706        assert_eq!(rtp.get_version(), RTP_VERSION);
707        assert_eq!(rtp.get_padding(), 0);
708        assert_eq!(rtp.get_extension(), 0);
709        assert_eq!(rtp.get_csrc_count(), 0);
710        assert_eq!(rtp.get_marker(), 0);
711        assert_eq!(rtp.get_payload_type(), RTP_PROFILE_TYPE);
712        assert_eq!(rtp.get_sequence(), sentinel_val.into());
713        assert_eq!(rtp.get_timestamp(), (sentinel_val as u32).into());
714        assert_eq!(rtp.get_ssrc(), sentinel_val as u32);
715    }

794    #[tokio::test]
795    async fn packet_blocks_are_cleaned_up() {
796        // Allocate 2 blocks.
797        let n_pkts = PACKETS_PER_BLOCK + 1;
798        let (mut sched, _listeners) = MockScheduler::from_mixers(
799            None,
800            (0..n_pkts)
801                .map(|_| Mixer::test_with_float(1, Handle::current(), false))
802                .collect(),
803        );
804
805        // Assert no cleanup at start.
806        assert!(sched.core.run_once());
807        assert_eq!(sched.core.needed_blocks(), 2);
808        assert!(sched.core.excess_buffer_cull_time.is_none());
809
810        // Remove only entry in last block. Cleanup should be sched'd.
811        sched.core.remove_task(n_pkts - 1);
812        assert!(sched.core.run_once());
813        assert!(sched.core.has_excess_blocks());
814        assert!(sched.core.excess_buffer_cull_time.is_some());
815
816        tokio::time::sleep(Duration::from_secs(2) + MEMORY_CULL_TIMER).await;
817
818        // Cleanup should be unsched'd.
819        assert!(sched.core.run_once());
820        assert!(sched.core.excess_buffer_cull_time.is_none());
821        assert!(!sched.core.has_excess_blocks());
822    }
