1use std::time::Duration;
2
3use rand::random;
4
5use crate::FloatDuration;

7/// Logic used to determine how long to wait between retry attempts.
8#[derive(Clone, Copy, Debug, PartialEq)]
9#[non_exhaustive]
10pub enum Strategy {
11    /// The driver will wait for the same amount of time between each retry.
12    Every(FloatDuration),
13    /// Exponential backoff waiting strategy, where the duration between
14    /// attempts (approximately) doubles each time.
15    Backoff(ExponentialBackoff),
16}
17
18impl Strategy {
19    pub(crate) fn retry_in(&self, last_wait: Option<FloatDuration>) -> FloatDuration {
20        match self {
21            Self::Every(t) => *t,
22            Self::Backoff(exp) => exp.retry_in(last_wait),
23        }
24    }
25}
26
27/// Exponential backoff waiting strategy.
28///
29/// Each attempt waits for twice the last delay plus/minus a
30/// random jitter, clamped to a min and max value.
31#[derive(Clone, Copy, Debug, PartialEq)]
32pub struct ExponentialBackoff {
33    /// Minimum amount of time to wait between retries.
34    ///
35    /// *Defaults to 0.25s.*
36    pub min: FloatDuration,
37    /// Maximum amount of time to wait between retries.
38    ///
39    /// This will be clamped to `>=` min.
40    ///
41    /// *Defaults to 10s.*
42    pub max: FloatDuration,
43    /// Amount of uniform random jitter to apply to generated wait times.
44    /// I.e., 0.1 will add +/-10% to generated intervals.
45    ///
46    /// This is restricted to within +/-100%.
47    ///
48    /// *Defaults to `0.1`.*
49    pub jitter: f32,
50}
51
52impl Default for ExponentialBackoff {
53    fn default() -> Self {
54        Self {
55            min: Duration::from_millis(250).into(),
56            max: Duration::from_secs(10).into(),
57            jitter: 0.1,
58        }
59    }
60}
61
62impl ExponentialBackoff {
63    pub(crate) fn retry_in(&self, last_wait: Option<FloatDuration>) -> FloatDuration {
64        let min = self.min.as_secs_f32();
65        let max = self.max.as_secs_f32();
66
67        let attempt = last_wait.map_or(min, |t| 2.0 * t.as_secs_f32());
68        let perturb = (1.0 - (self.jitter * 2.0 * (random::<f32>() - 1.0))).clamp(0.0, 2.0);
69        let mut target_time = attempt * perturb;
70
71        // Now clamp target time into given range.
72        let safe_max = if max < min { min } else { max };
73        target_time = target_time.clamp(min, safe_max);
74
75        FloatDuration::from_secs_f32(target_time)
76    }
77}
