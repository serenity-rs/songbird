17/// Opus decoder for symphonia, based on libopus v1.5 (via [`opus2`]).
18pub struct OpusDecoder {
19    inner: Opus2Decoder,
20    params: CodecParameters,
21    buf: AudioBuffer<f32>,
22    rawbuf: Vec<f32>,
23}
24
25/// # SAFETY
26/// The underlying Opus decoder (currently) requires only a `&self` parameter
27/// to decode given packets, which is likely a mistaken decision.
28///
29/// This struct makes stronger assumptions and only touches FFI decoder state with a
30/// `&mut self`, preventing data races via `&OpusDecoder` as required by `impl Sync`.
31/// No access to other internal state relies on unsafety or crosses FFI.
32unsafe impl Sync for OpusDecoder {}
33
34impl OpusDecoder {
35    fn decode_inner(&mut self, packet: &Packet) -> SymphResult<()> {
36        let s_ct = loop {
37            if packet.buf().len() > i32::MAX as usize {
38                return decode_error("Opus packet was too large (greater than i32::MAX bytes).");
39            }
40
41            match self
42                .inner
43                .decode_float(packet.buf(), &mut self.rawbuf, false)
44            {
45                Ok(v) => break v,
46                Err(e) if e.code() == ErrorCode::BufferTooSmall => {
47                    // double the buffer size
48                    // correct behav would be to mirror the decoder logic in the udp_rx set.
49                    let new_size = (self.rawbuf.len() * 2).min(i32::MAX as usize);
50                    if new_size == self.rawbuf.len() {
51                        return decode_error("Opus frame too big: cannot expand opus frame decode buffer any further.");
52                    }
53
54                    self.rawbuf.resize(new_size, 0.0);
55                    self.buf = AudioBuffer::new(
56                        self.rawbuf.len() as u64 / 2,
57                        SignalSpec::new_with_layout(SAMPLE_RATE_RAW as u32, Layout::Stereo),
58                    );
59                },
60                Err(e) => {
61                    tracing::error!("Opus decode error: {:?}", e);
62                    return decode_error("Opus decode error: see 'tracing' logs.");
63                },
64            }
65        };
66
67        self.buf.clear();
68        self.buf.render_reserved(Some(s_ct));
69
70        // Forcibly assuming stereo, for now.
71        for ch in 0..2 {
72            let iter = self.rawbuf.chunks_exact(2).map(|chunk| chunk[ch]);
73            for (tgt, src) in self.buf.chan_mut(ch).iter_mut().zip(iter) {
74                *tgt = src;
75            }
76        }
77
78        Ok(())
79    }
80}

82impl Decoder for OpusDecoder {
83    fn try_new(params: &CodecParameters, _options: &DecoderOptions) -> SymphResult<Self> {
84        let inner = Opus2Decoder::new(SAMPLE_RATE, Channels::Stereo).unwrap();
85
86        let mut params = params.clone();
87        params.with_sample_rate(SAMPLE_RATE_RAW as u32);
88
89        Ok(Self {
90            inner,
91            params,
92            buf: AudioBuffer::new(
93                MONO_FRAME_SIZE as u64,
94                SignalSpec::new_with_layout(SAMPLE_RATE_RAW as u32, Layout::Stereo),
95            ),
96            rawbuf: vec![0.0f32; STEREO_FRAME_SIZE],
97        })
98    }
99
100    fn supported_codecs() -> &'static [CodecDescriptor] {
101        &[symphonia_core::support_codec!(
102            CODEC_TYPE_OPUS,
103            "opus",
104            "libopus (1.5+, opus2)"
105        )]
106    }
107
108    fn codec_params(&self) -> &CodecParameters {
109        &self.params
110    }
111
112    fn decode(&mut self, packet: &Packet) -> SymphResult<AudioBufferRef<'_>> {
113        if let Err(e) = self.decode_inner(packet) {
114            self.buf.clear();
115            Err(e)
116        } else {
117            Ok(self.buf.as_audio_buffer_ref())
118        }
119    }
120
121    fn reset(&mut self) {
122        _ = self.inner.reset_state();
123    }
124
125    fn finalize(&mut self) -> FinalizeResult {
126        FinalizeResult::default()
127    }
128
129    fn last_decoded(&self) -> AudioBufferRef<'_> {
130        self.buf.as_audio_buffer_ref()
131    }
132}
