11/// Storage for [`EventData`], designed to be used for both local and global contexts.
12///
13/// Timed events are stored in a binary heap for fast selection, and have custom `Eq`,
14/// `Ord`, etc. implementations to support (only) this.
15///
16/// [`EventData`]: EventData
17pub struct EventStore {
18    timed: BinaryHeap<EventData>,
19    untimed: Box<EnumMap<UntimedEvent, Vec<EventData>>>,
20    local_only: bool,
21}
22
23impl EventStore {
24    /// Creates a new event store to be used globally.
25    #[must_use]
26    pub fn new() -> Self {
27        Self::default()
28    }
29
30    /// Creates a new event store to be used within a [`Track`].
31    ///
32    /// This is usually automatically installed by the driver once
33    /// a track has been registered.
34    ///
35    /// [`Track`]: crate::tracks::Track
36    #[must_use]
37    pub fn new_local() -> Self {
38        EventStore {
39            local_only: true,
40            ..Default::default()
41        }
42    }

44    /// Add an event to this store.
45    ///
46    /// Updates `evt` according to [`EventData::compute_activation`].
47    ///
48    /// [`EventData::compute_activation`]: EventData::compute_activation
49    pub fn add_event(&mut self, mut evt: EventData, now: Duration) {
50        evt.compute_activation(now);
51
52        if self.local_only && evt.event.is_global_only() {
53            return;
54        }
55
56        match evt.event {
57            Event::Core(c) => {
58                self.untimed[c.into()].push(evt);
59            },
60            Event::Track(t) => {
61                self.untimed[t.into()].push(evt);
62            },
63            Event::Delayed(_) | Event::Periodic(_, _) => {
64                self.timed.push(evt);
65            },
66            _ => {
67                // Event cancelled.
68            },
69        }
70    }
71
72    /// Processes all events due up to and including `now`.
73    pub(crate) async fn process_timed(&mut self, now: Duration, ctx: EventContext<'_>) {
74        while let Some(evt) = self.timed.peek() {
75            if evt
76                .fire_time
77                .as_ref()
78                .expect("Timed event must have a fire_time.")
79                > &now
80            {
81                break;
82            }
83            let mut evt = self
84                .timed
85                .pop()
86                .expect("Can only succeed due to peek = Some(...).");
87
88            let old_evt_type = evt.event;
89            if let Some(new_evt_type) = evt.action.act(&ctx).await {
90                evt.event = new_evt_type;
91                self.add_event(evt, now);
92            } else if let Event::Periodic(d, _) = old_evt_type {
93                evt.event = Event::Periodic(d, None);
94                self.add_event(evt, now);
95            }
96        }
97    }
98
99    /// Processes all events due up to and including `now`.
100    pub(crate) fn timed_event_ready(&self, now: Duration) -> bool {
101        self.timed.peek().is_some_and(|evt| {
102            evt.fire_time
103                .as_ref()
104                .expect("Timed event must have a fire_time.")
105                <= &now
106        })
107    }
108
109    /// Processes all events attached to the given track event.
110    pub(crate) async fn process_untimed(
111        &mut self,
112        now: Duration,
113        untimed_event: UntimedEvent,
114        ctx: EventContext<'_>,
115    ) {
116        // move a Vec in and out: not too expensive, but could be better.
117        // Although it's obvious that moving an event out of one vec and into
118        // another necessitates that they be different event types, thus entries,
119        // convincing the compiler of this is non-trivial without making them dedicated
120        // fields.
121        let mut events = std::mem::take(&mut self.untimed[untimed_event]);
122        if !events.is_empty() {
123            // TODO: Possibly use tombstones to prevent realloc/memcpys?
124            // i.e., never shrink array, replace ended tracks with <DEAD>,
125            // maintain a "first-track" stack and freelist alongside.
126            let mut i = 0;
127            while i < events.len() {
128                let evt = &mut events[i];
129                // Only remove/readd if the event type changes (i.e., Some AND new != old)
130                if let Some(new_evt_type) = evt.action.act(&ctx).await {
131                    if evt.event == new_evt_type {
132                        let mut evt = events.remove(i);
133
134                        evt.event = new_evt_type;
135                        self.add_event(evt, now);
136                    } else {
137                        i += 1;
138                    }
139                } else {
140                    i += 1;
141                };
142            }
143            self.untimed[untimed_event] = events;
144        }
145    }
146}

148#[derive(Debug, Default)]
149pub(crate) struct GlobalEvents {
150    pub(crate) store: EventStore,
151    pub(crate) time: Duration,
152    pub(crate) awaiting_tick: HashMap<TrackEvent, Vec<usize>>,
153}
154
155impl GlobalEvents {
156    pub(crate) fn add_event(&mut self, evt: EventData) {
157        self.store.add_event(evt, self.time);
158    }
159
160    pub(crate) async fn fire_core_event(&mut self, evt: CoreEvent, ctx: EventContext<'_>) {
161        self.store.process_untimed(self.time, evt.into(), ctx).await;
162    }
163
164    pub(crate) fn fire_track_event(&mut self, evt: TrackEvent, index: usize) {
165        let holder = self.awaiting_tick.entry(evt).or_default();
166
167        holder.push(index);
168    }
169
170    pub(crate) fn remove_handlers(&mut self) {
171        self.store = EventStore::new();
172    }
173
174    pub(crate) async fn tick(
175        &mut self,
176        events: &mut [EventStore],
177        states: &mut [TrackState],
178        handles: &mut [TrackHandle],
179    ) {
180        // Global timed events
181        self.time += TIMESTEP_LENGTH;
182        if self.store.timed_event_ready(self.time) {
183            let global_ctx: Vec<(&TrackState, &TrackHandle)> =
184                states.iter().zip(handles.iter()).collect();
185            self.store
186                .process_timed(self.time, EventContext::Track(&global_ctx[..]))
187                .await;
188        }
189
190        // Local timed events
191        for (i, state) in states.iter_mut().enumerate() {
192            if state.playing.is_playing() && state.ready == ReadyState::Playable {
193                state.step_frame();
194
195                let event_store = events
196                    .get_mut(i)
197                    .expect("Missing store index for Tick (local timed).");
198                let handle = handles
199                    .get_mut(i)
200                    .expect("Missing handle index for Tick (local timed).");
201
202                event_store
203                    .process_timed(state.play_time, EventContext::Track(&[(state, handle)]))
204                    .await;
205            }
206        }
207
208        for (evt, indices) in &self.awaiting_tick {
209            let untimed = (*evt).into();
210
211            if !indices.is_empty() {
212                info!("Firing {:?} for {:?}", evt, indices);
213            }
214
215            // Local untimed track events.
216            for &i in indices {
217                let event_store = events
218                    .get_mut(i)
219                    .expect("Missing store index for Tick (local untimed).");
220                let handle = handles
221                    .get_mut(i)
222                    .expect("Missing handle index for Tick (local untimed).");
223                let state = states
224                    .get_mut(i)
225                    .expect("Missing state index for Tick (local untimed).");
226
227                event_store
228                    .process_untimed(
229                        state.position,
230                        untimed,
231                        EventContext::Track(&[(state, handle)]),
232                    )
233                    .await;
234            }
235
236            // Global untimed track events.
237            if !self.store.untimed[untimed].is_empty() && !indices.is_empty() {
238                let global_ctx: Vec<(&TrackState, &TrackHandle)> = indices
239                    .iter()
240                    .map(|i| {
241                        (
242                            states
243                                .get(*i)
244                                .expect("Missing state index for Tick (global untimed)"),
245                            handles
246                                .get(*i)
247                                .expect("Missing handle index for Tick (global untimed)"),
248                        )
249                    })
250                    .collect();
251
252                self.store
253                    .process_untimed(self.time, untimed, EventContext::Track(&global_ctx[..]))
254                    .await;
255            }
256        }
257
258        // Now drain vecs.
259        for indices in self.awaiting_tick.values_mut() {
260            indices.clear();
261        }
262    }
